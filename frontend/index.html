🚀 Deploy commands

cd backend && pip install -r requirements.txt
uvicorn app.main:app --host 0.0.0.0 --port 8000      # local (mock mode)

gcloud run deploy socialauto --source . --region asia-south1 --allow-unauthenticated \
  --set-env-vars MOCK_MODE=false,CRON_SECRET=your_long_secret
# then add GitHub secrets: APP_URL + CRON_SECRET  → posts auto-publish every 15 min
